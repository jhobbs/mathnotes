\section{Linear Transformations}

\begin{definition}[Linear Transformation]\synonyms{"linear"}
	Let $V$ and $W$ be \@{vector spaces} over a \@{field} $\mathbb{F}.$ We call a \@{function}

	\[ T: V \to W \] a \term{linear transformation} from $V$ to $W$ if for all $\vec{x}, \vec{y} \in V$ and all $c \in \mathbb{F},$ we have

	\begin{enumerate}[label=(\alph*)]
		\item $T(x + y) = T(x) + T(y)$ and
		\item $T(cx) = cT(x)$
	\end{enumerate}

	The first condition is called \term{additivity} and the second is called \term{homogeniety}.
\end{definition}

\begin{definition}[Null Space]
	Let $V$ and $W$ be \@{vector spaces} and let $T: V \to W$ be \@{linear}.

	We define the \term[kernel]{null space} $N(T)$ of $T$ to be the \@{set} of all \@{vectors} $x \in V$ such that $T(x) = 0,$ that is, $N(T) = \{ \vec{x} \in V : T(\vec{x}) = 0. \}$

	We define the \term[image of a linear transformation]{range of a linear transformation} $R(T)$ of $T$ to be the \@{subset} of $W$ consisting of all \@{images} (under T) of vectors in $V,$ that is, $R(T) = \{T(\vec{x}) : \vec{x} \in V\}.$
\end{definition}

\begin{theorem}[Null Space and Range are Subspaces]
	Let $V$ and $W$ be \@{vector spaces} and let $T: V \to W$ be \@{linear}. Then the \@{null space} and \@[range]{range of a linear transformation} of $T$ are \@{subspaces} of $V$ and $W,$ respectively.
\end{theorem}

\begin{theorem}[Range is span of transformation of domain basis]
	Let $V$ and $W$ be \@{vector spaces} and let $T: V \to W$ be \@{linear}. If $\beta = \{ v_1, v_2, \dots, v_n \}$ is a \@{basis} for $V,$ then

	\[ R(T) = \text{span}{(T(\beta))} = \text{span}{(\{T(v_1), T(v_2), \dots, T(v_n) \} )}. \]
\end{theorem}

\begin{definition}[rank]
\notation{\rank}{\operatorname{rank}}
	Let $V$ and $W$ be \@{vector spaces} and let $T: V \to W$ be \@{linear}. If $N(T)$ and $R(T)$ (\@{null space} and \@[range]{range of a linear transformation} of $T$, respectively) are \@{finite dimensional}, the we define the \term{nullity} of $T,$ denoted $\text{nullity}(T),$ and the \term{rank} of $T,$ denoted $\text{rank}(T),$ to be the \@{dimensions} of $N(T)$ and $R(T),$ respectively.
\end{definition}

\begin{theorem}[Rank Nullity Theorem]
	Let $V$ and $W$ be \@{vector spaces} and let $T: V \to W$ be \@{linear}. If $V$ is \@{finite-dimensional}, then

	\[ \text{rank}{(T)} + \text{nullity}{(T)} = \text{dim}{(V)}. \]
\end{theorem}
