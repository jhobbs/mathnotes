\section{Euler's Method}

We'll consider a first order IVP of the form

\[ \frac{du}{dt} f(t, u), \quad u(0) = c, \]

with the goal of numerically estimating the solution on a finite time interval $[0, T]$ for some $T > 0.$

First we'll establish this is a reasonable thing to do.

\begin{theorem}[Well-posedness of IVPs]
Consider the IVP

	\[ \frac{d\vec{u}}{dt} = \vec{f}(t, \vec{u}), \quad \vec{u}(0) = \vec{c}. \]

	Suppose that $\vec{f}: [0, T] \times \reals^d \to \reals^d$ is \@{continuous} in its \@{domain} and \@{Lipschitz continuous} in $\vec{u}$, i.e., there exists a constant $L > 0$ such that

	\[ \abs{\vec{f}(t, \vec{u}) - \vec{f}(t, \vec{v})} \leq L\norm{\vec{u} - \vec{v}}, \forall(t, \vec{u}), (t, \vec{v}) \in [0, T] \times \reals^d. \]

	Then it holds that:

	\begin{enumerate}
	        \item \textbf{Existence and Uniqueness:} There exists a unique solution $\vec{u}(t)$ for any $\vec{u}_0 \in \reals^d.$
		\item \textbf{Continuous Dependence on initial data and right hand side:} Suppose $\vec{r} : [0,T] \times \reals^d \to \reals^d$ is a bounded \@{perturbation} of $\vec{f},$ i.e. there exists a constant $M > 0$ such that
			\[ \norm{\vec{r}(t, \vec{u})} \leq M, \quad \forall(t, \vec{u}) \in [0, T] \times \reals^d. \]
			Further let $\vec{\hat{u}}(t)$ be the solution of the perturbed IVP

			\[ \frac{d \vec{\hat{u}}}{dt} = \vec{f}(t, \vec{\hat{u}}) + \vec{r}(t, \vec{\hat{u}}), \quad \vec{\hat{u}}(0) = \vec{\hat{c}},  \]
			which we assume exists and is unique. Then we have

			\[ \norm{\vec{u}(t) - \vec{\hat{u}}(t)} \leq e^{Lt} \norm{\vec{c} - \vec{\hat{c}}} + \frac{M}{L}(e^{Lt} - 1), \quad \forall t \in [0, T]. \]
	\end{enumerate}

\end{theorem}

Now that we've established that IVPs are well posed, we can look at how to approximate our unknown function $u(t)$ at a discrete set of points in time. 

\begin{definition}[discrete time points]\synonyms{"time mesh points", "time grid points"}
	Consider a \@{partition} $0 = t_0 < t_1 < \cdots < t_N = T$ of the time interval $[0, T].$ We can call the points $t_n$ the \textbf{discrete time points}. Then our goal is to approximate the values $u(t_n)$ with a set of quantities $\set{u_n}_{n=0}^N,$ that is, $u_n \approx u(t_n)$ for $n = 0, 1, \dots, N.$ We can also write $\delta t_n = t_{n+1} - t_n$ to denote the \term[time step]{step size} between between consecutive points. For simplicity we often assume that $\delta t_n = \delta t$ for all $n = 0, 1, \dots, N-1,$ i.e. the time steps are constant. Such a mesh is called a \@{uniform mesh}.
\end{definition}

\begin{definition}[I.V.P. solution approximation]
	Now we can talk about \textbf{I.V.P. solution approximations}
\[ \vec{u}_0 (=\vec{c}), \vec{u}_1, \dots, \vec{u}_{N-1}, \vec{u}_N \]
	where $\vec{u}_n$ is an intended \@{approximation} of $\vec{u}(t_n).$
\end{definition}

When we have an \@{I.V.P}, we know $\vec{u}_0 = \vec{c}$ and can integrate the ODE in steps, where on each step $n (1 \leq n \leq N)$ we know an approximation $\vec{u}_{n-1}$ at $t_{n-1}$ and we seek $\vec{u}_n$ at $t_n$. Thus, we don't need advanced knowledge of the entire \@{mesh} beyond it. Let's look at one such step, $n (\geq 1)$. But first, a little more on order.

\begin{definition}[Big-O order]\synonyms{"big-O notation"}
	\notation{\bigO}{\mathcal{O}\!\left(#1\right)}
	Given a \@{step size} $h > 0$, we denote for a \@{vector} $\vec{d}$ depending on $h$
	\[ \vec{d} = \bigO{h^p} \]
	if there exist positive constants $p$ and $C$ such that for all $h > 0$ small enough,
	\[ \norm{\vec{d}} \leq Ch^p. \]
	We say $\vec{d}$ is \term{big-O of $h^p$}, written $\vec{d} = \bigO{h^p}$.
\end{definition}

\begin{definition}[Forward Euler Method]\synonyms{"forward euler method", "explicit euler", "explicit euler method", "euler's method"}
	Given an \@{I.V.P}
	
	\[ \frac{d\vec{u}}{dt} = \vec{f}(t, \vec{u}), \quad \vec{u}(0) = \vec{c} \]

	and a \@{time mesh} with step size $h_n,$ We'll construct a \@{discretization} method by considering the \@{Taylor expansion} of $\vec{u}(t_n)$ about $t_{n-1}$:
\[ \vec{u}(t_n) =  \vec{u}(t_{n-1}) + h_n \vec{u}'(t_{n-1}) + \frac{1}{2}{h_n}^2 \vec{y}''(t_{n-1}) + \cdots \]
which we can also use \@{big-O notation} to write as
\[ \vec{u}(t_n) =  \vec{u}(t_{n-1}) + h_n \vec{u}'(t_{n-1}) + \bigO{{h_n}^2}. \]
	The \textbf{forward Euler method} can be derived by dropping the rightmost term in this Taylor expansion and replacing $\vec{u}'$ with $\vec{f},$ yielding the scheme for \@{I.V.P. solution approximation}
	\[ \vec{u}_n = \vec{u}_{n-1} + h_n \vec{f}(t_{n-1}, \vec{u}_{n-1}). \]
\end{definition}
\begin{intuition}
	We're given the derivative of our solution ($\vec{f}$) as part of the $I.V.P.$; this method works by, starting at $u_{n-1},$ taking a step of length $h_n$ in the tangential direction (given by the $\vec{f}(t_{n-1}, u_{n-1})$ and hoping that with small enough $h_n$, we wind up somewhere near $\vec{u}(t_n)$. In the next session, we'll assess this hope.
\end{intuition}

\section{Convergence, Accuracy, Consistency and 0-Stability}

We'll rewrite \@{euler's method} in a form compatible with our approximated ODE:

\[ \frac{\vec{u}_n - \vec{u}_{n-1}}{h_n} - \vec{f}(t_{n-1}, \vec{y}_{n-1}) = \vec{0}. \]

We'll formalize a bit with the following definition.

\begin{definition}[Difference Operator]
	\notation{\difference}{\mathscr{N}_h}
	Define the \term{difference operator} $\difference$ by
	\[ \difference \vec{v}(t_n) \equiv \frac{\vec{v}(t_n) - \vec{v}(t_{n-1})}{h_n} - \vec{f}(t_{n-1}, \vec{v}_{n-1}), \]
	applied for $n = 1,2,\dots,N$ for any \@{function} $\vec{v}$ defined at \@{mesh points} with $\vec{v}(t_0)$ specified.
\end{definition}

Consider $\vec{u}_h$ to be a \@{mesh function} which takes on the value $\vec{h}_n$ at each $t_n.$  Then the numerical method is given by
\[ \difference \vec{u}_h(t_n) = 0, \]
with $\vec{u}_0 = \vec{c}.$

Much of the study of numerical ODEs is concerned with the errors on each step that are due to difference approximation, and how they accumulate across steps. One measure of the error made on each step follows.

\begin{definition}[Local truncation error]
	The \@{residual} of the \@{difference operator} when it is applied to the exact solution $\vec{u}(t_n)$ is called the \textbf{local truncation error} and is given by
	\[ \vec{\tau}_n = \difference \vec{u}(t_n). \]

	This is a measure of how closely the \@{difference operator} approximates the \@{differential operator}, and applies to other discretization methods beyond \@{euler's method} as well, differing in the definition of the difference operator.
\end{definition}

\begin{definition}[order $n$ accurate]\synonyms{"order n consistent", "accurate", "consistent"}
	The \@{difference method} (approximation scheme) is said to be \textbf{order} $p$ \textbf{accurate} if

	\[ \vec{\tau}_n = \bigO{{h_n}^p} \]

	for a positive integer $p$.
\end{definition}

\begin{theorem}[\@{forward euler method} is order $1$ \@{accurate}]
\end{theorem}
\begin{proof}
For the \@{forward euler method}, we have from the definition of the \@{difference operator}
\bal
	\vec{\tau}_n & = \difference \vec{u}(t_n) \\
	     & = \frac{\vec{u}(t_n) - \vec{u}(t_{n-1})}{h_n} - \vec{f}(t_{n-1}, \vec{u}_{n-1}). \\
	\shortintertext{Now we replace $\vec{u}$ with our taylor expansion given above, and $\vec{f}(t_{n-1}, \vec{u}_{n-1})$ with $\vec{u}'(t_{n-1})$:}
	     & = \frac{\vec{u}(t_{n-1}) + h_n \vec{u}'(t_{n-1}) + \bigO{{h_n}^2} - \vec{u}(t_{n-1})}{h_n} - \vec{u}'(t_{n-1}) \\
	     & = \frac{h_n \vec{u}'(t_{n-1}) + \bigO{{h_n}^2}}{h_n} - \vec{u}'(t_{n-1}) \\
	     & = \vec{u}'(t_{n-1}) + \bigO{h_n} - \vec{u}'(t_{n-1}) \\
	     & = \bigO{h_n}.
\eal
\end{proof}

\section{Backward Euler}
For the \@{forward euler method} we expanded $\vec{u}(t_n)$ at $t_{n-1}$ but we could have expanded it at another point and still had a valid Taylor expansion. We called it "forward" because we're taking the derivative at $t_{n-1}$ and using it to approximate the slope of $u$ moving forward to $t_{n}.$ Backward Euler is a related method that centers its \@{taylor expansion} at $t_n,$ which means we're using the tanget of a future point instead of a past point.

\begin{definition}[Backward Euler method]\synonyms{"backward euler"}
\textbf{Backward Euler} is just like \@{forward euler} except it uses an expansion centered at $t_n$ instead of $t_{n-1}$. It is given by the scheme
	\[ \vec{u}_n = \vec{u}_{n-1} + h_n \vec{f}(t_n, \vec{u}_n). \]
\end{definition}
\begin{note}
	\@{Backward euler} is an implicit method - notice that $\vec{u}_n$ is on both sides of the equation! This means we generally have to solve a nonlinear equation in order to use this method.
\end{note}


\begin{theorem}[The \@{backward euler method} is order $1$ \@{accurate}]
\end{theorem}
\begin{proof}
	We'll need the expansion of $\vec{u}(t_{n-1})$ at $t_n$:
	\[ \vec{u}(t_{n-1}) = \vec{u}(t_n) - h_{n-1}\vec{u}'(t_n) + O({h_{n-1}}^2). \]

	Now we'll define our \@{difference operator} as
	\[ \difference \vec{u}(t_n) \equiv \frac{\vec{u}(t_n) - \vec{u}(t_{n-1})}{h_{n-1}} - \vec{f}(t_n, \vec{u}_n). \]

	Then
	\bal
	\vec{\tau}_n & = \difference \vec{u}(t_n) \\
		     & = \frac{\vec{u}(t_n) - \vec{u}(t_{n-1})}{h_{n-1}} - \vec{f}(t_n, \vec{u}_n) \\
	\shortintertext{Substituting in our expansion and $u'(t_n)$ for $\vec{f}(t_n, \vec{u}_n)$ gives}
		     & = \frac{\vec{u}(t_n) -  \vec{u}(t_n) + h_{n-1}\vec{u}'(t_n) + O({h_{n-1}}^2)}{h_{n-1}} - \vec{u}'(t_n) \\
		     & = \frac{h_{n-1}\vec{u}'(t_n) + O({h_{n-1}}^2)}{h_{n-1}} - u'(t_n) \\
		     & = \vec{u}'(t_n) + O(h_{n-1})  - \vec{u}'(t_n) \\
		     & = O(h_{n-1}). \\
	\eal
\end{proof}
